# 假光标

CDP 不采集系统光标，录制里没有鼠标，必须往页面注入假光标。一次注入，跨导航有效。

```js
const install = () => {
  if (document.getElementById('__sej-cursor')) return;
  const s = document.createElement('style');
  s.textContent = 'html.__sej-nc, html.__sej-nc *{cursor:none !important}';
  document.documentElement.append(s);
  document.documentElement.classList.add('__sej-nc');
  const el = document.createElement('div');
  el.id = '__sej-cursor';
  el.style.cssText = 'position:fixed;left:0;top:0;width:24px;height:24px;z-index:2147483647;pointer-events:none;will-change:transform;transform:translate3d(-200px,-200px,0);';
  el.innerHTML = '<svg width="24" height="24" viewBox="0 0 24 24"><path d="M4.5 2.5 L4.5 19.2 L9.1 14.8 L12.2 21.3 L15 20 L11.9 13.6 L18.2 13Z" fill="#fff" stroke="#202124" stroke-width="1.5" stroke-linejoin="round" style="filter:drop-shadow(0 1px 3px rgba(0,0,0,.5))"/></svg>';
  document.documentElement.append(el);
  const ring = document.createElement('div');
  ring.style.cssText = 'position:fixed;width:36px;height:36px;margin:-18px 0 0 -18px;border-radius:50%;border:2px solid rgba(26,115,232,.85);background:rgba(26,115,232,.16);opacity:0;transform:scale(.4);transition:opacity .4s,transform .4s;z-index:2147483646;pointer-events:none;';
  document.documentElement.append(ring);
  const move = (x, y) => {
    el.style.transform = `translate3d(${x - 5}px, ${y - 2}px, 0)`;
    ring.style.left = x + 'px'; ring.style.top = y + 'px';
  };
  addEventListener('mousemove', (e) => move(e.clientX, e.clientY), true);
  addEventListener('mousedown', (e) => {
    move(e.clientX, e.clientY);
    ring.style.transition = 'none'; ring.style.opacity = '1'; ring.style.transform = 'scale(.45)';
    requestAnimationFrame(() => {
      ring.style.transition = 'opacity .5s,transform .5s';
      ring.style.opacity = '0'; ring.style.transform = 'scale(1.2)';
    });
  }, true);
};
await page.addInitScript({ content: `(${install.toString()})()` });  // 跨导航
await page.evaluate(install);                                        // 当前页
await page.mouse.move(700, 500, { steps: 5 });                       // 触发首次定位
```
